<?php

declare(strict_types=1);

namespace App\Enums;

use App\Attributes\Colour;
use App\Attributes\EnumMeta;
use App\Contracts\Labelled;
use Illuminate\Database\Eloquent\Model;

/**
 * Kitchen-sink backed enum
 *
 * @template TMeta
 */
#[EnumMeta(group: 'content')]
enum KitchenStatus: string implements Labelled, JsonSerializable
{
    #[Colour(value: 'grey')]
    case Draft = 'draft';
    case Published = 'published';
    case Archived = 'archived';

    private const self DEFAULT = self::Draft;

    public function label(): string
    {
        return match ($this) {
            self::Draft => 'Draft',
            self::Published => 'Published',
            self::Archived => 'Archived',
        };
    }

    /**
     * @throws ValueError
     */
    public static function fromModel(Model $model): self
    {
        return self::from((string) $model->status);
    }

    public function jsonSerialize(): array
    {
        return ["value" => $this->value];
    }
}
